        local percent = math.clamp((input.Position.X - trackPos) / trackSize, 0, 1)

ProximityPromptService.PromptButtonHoldBegan:Connect(function(prompt)
    if not Config.flashEnabled and not Config.giantEnabled then return end
    if activePrompts[prompt] then return end
    
    activePrompts[prompt] = true
    local startTime = os.clock()
    local fired = false
    
    ProgressFillBar.Size = UDim2.new(0, 0, 1, 0)
    BarPercent.Text = "0%"
    
    local connection
    connection = RunService.PreRender:Connect(function()
        if not prompt or not prompt.Parent then
            connection:Disconnect()
            activePrompts[prompt] = nil
            ProgressFillBar.Size = UDim2.new(0, 0, 1, 0)
            BarPercent.Text = "0%"
            return
        end
        
        local progress = math.clamp((os.clock() - startTime) / prompt.HoldDuration, 0, 1)
        local progressPercent = math.floor(progress * 100)
        
        ProgressFillBar.Size = UDim2.new(progress, 0, 1, 0)
        BarPercent.Text = progressPercent .. "%"
        
        if not fired and progress >= triggerPercent then
            fired = true
            connection:Disconnect()
            activePrompts[prompt] = nil
            activateFlashAndPotion()
            
            task.delay(0.25, function()
                TweenService:Create(ProgressFillBar, TweenInfo.new(0.35, Enum.EasingStyle.Quad), {
                    Size = UDim2.new(0, 0, 1, 0)
                }):Play()
                BarPercent.Text = "0%"
            end)
        end
    end)
    
    prompt.PromptButtonHoldEnded:Connect(function()
        if not fired then
            connection:Disconnect()
            activePrompts[prompt] = nil
            TweenService:Create(ProgressFillBar, TweenInfo.new(0.3), {
                Size = UDim2.new(0, 0, 1, 0)
            }):Play()
            BarPercent.Text = "0%"
        end

-- ==================== DRAGGING FÜR MAIN PANEL (MOBILE KOMPATIBEL) ====================
local DragHandle = Instance.new("Frame")
DragHandle.Name = "DragHandle"
DragHandle.Size = UDim2.new(1, 0, 0, 40)
DragHandle.BackgroundTransparency = 1
DragHandle.Parent = MainWindow

local dragStartPos = nil
local dragStartMouse = nil
local windowDragging = false
local dragConnection = nil

DragHandle.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        windowDragging = true
        dragStartMouse = UserInputService:GetMouseLocation()
        dragStartPos = MainWindow.Position
        
        if dragConnection then dragConnection:Disconnect() end
        
        dragConnection = RunService.RenderStepped:Connect(function()
            if windowDragging then
                local mousePos = UserInputService:GetMouseLocation()
                local delta = mousePos - dragStartMouse
                MainWindow.Position = UDim2.new(
                    dragStartPos.X.Scale,
                    dragStartPos.X.Offset + delta.X,
                    dragStartPos.Y.Scale,
                    dragStartPos.Y.Offset + delta.Y
                )
            end
        end)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        windowDragging = false
        if dragConnection then
            dragConnection:Disconnect()
            dragConnection = nil
        end
    end
end)

    MainScreen:Destroy()
    if TpScreen and TpScreen.Parent then

    end
end)
